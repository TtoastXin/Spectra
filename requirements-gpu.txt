# GPU additions to requirements.txt, for the experiment scripts on an NVIDIA GPU.
# Install the CPU build of PyTorch first (it is only used, on the CPU, for the
# C2ST and MMTV metrics; its CUDA build needs cuDNN 9, which conflicts with the
# cuDNN 8.9 of jax[cuda12]==0.4.28), then both requirement files:
#
#     pip install torch==2.6.0 --index-url https://download.pytorch.org/whl/cpu
#     pip install -r requirements.txt -r requirements-gpu.txt
#
# The CUDA libraries are pinned to the versions used for the paper's GPU runs.
# jax[cuda12] alone only bounds them from below, so pip would otherwise install
# newer ones.
jax[cuda12]==0.4.28
nvidia-cublas-cu12==12.4.5.8
nvidia-cuda-cupti-cu12==12.4.127
nvidia-cuda-nvcc-cu12==12.9.86
nvidia-cuda-nvrtc-cu12==12.9.86
nvidia-cuda-runtime-cu12==12.4.127
nvidia-cudnn-cu12==8.9.7.29
nvidia-cufft-cu12==11.2.1.3
nvidia-cusolver-cu12==11.6.1.9
nvidia-cusparse-cu12==12.3.1.170
nvidia-nccl-cu12==2.21.5
nvidia-nvjitlink-cu12==12.4.127
